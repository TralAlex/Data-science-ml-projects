
import numpy as np
import matplotlib.pyplot as plt
from datetime import datetime, timedelta

# K-MEANS mapa podeljena na 3 dela po: Datum (dan) & Cena
np.random.seed(42)

# 0) Tabela: Number, Date (20–25.12.2025), Party, Cena (1–1000)
partije = np.array(['Rainbow','Curry','Ketchup','Icecream'])
start_date = datetime(2025, 12, 20)
end_date   = datetime(2025, 12, 25)

n_per_day = 20
num_days = (end_date - start_date).days + 1
n = n_per_day * num_days

incident_numbers = np.random.randint(10000, 20000, size=n)
dates = np.array([start_date + timedelta(days=d) for d in range(num_days) for _ in range(n_per_day)])
party = np.random.choice(partije, size=n, replace=True)

day_idx = np.array([dt.day for dt in dates])  # 20..25
base_price = np.random.randint(1, 1001, size=n)
price = np.clip(base_price + (day_idx - 22) * 40 + np.random.randint(-30, 30, size=n), 1, 1000)

# 1) Feature-i za klasterisanje: X = [day, price]
X = np.c_[day_idx.astype(float), price.astype(float)]  # (n,2)

# 2) K-Means (sklearn ako postoji; inače fallback)
use_sklearn = True
try:
    from sklearn.cluster import KMeans
    from sklearn.preprocessing import StandardScaler
    from sklearn.pipeline import make_pipeline
except Exception:
    use_sklearn = False

k = 3           # mapa podeljena na 3 dela
scale = True    # standardizacija

if use_sklearn:
    model = make_pipeline(StandardScaler(), KMeans(n_clusters=k, init='k-means++', n_init=10, random_state=42))
    model.fit(X)
    labels = model.predict(X)
    scaler = model.named_steps['standardscaler']
    centers = scaler.inverse_transform(model.named_steps['kmeans'].cluster_centers_)
    def predict_grid(grid): return model.predict(grid)
else:
    mu = X.mean(axis=0); sigma = X.std(axis=0, ddof=0) + 1e-9
    Xs = (X - mu) / sigma
    rnd = np.random.RandomState(42)
    centers_s = [Xs[rnd.randint(0, Xs.shape[0])]]
    for _ in range(1, k):
        d2 = np.min([np.sum((Xs - c)**2, axis=1) for c in centers_s], axis=0)
        probs = d2 / (d2.sum() + 1e-12)
        centers_s.append(Xs[rnd.choice(Xs.shape[0], p=probs)])
    centers_s = np.vstack(centers_s)
    labels = np.zeros(Xs.shape[0], dtype=int)
    for _ in range(300):
        dists = np.stack([np.sum((Xs - centers_s[j])**2, axis=1) for j in range(k)], axis=1)
        new_labels = np.argmin(dists, axis=1)
        new_centers = centers_s.copy()
        for j in range(k):
            m = (new_labels == j)
            new_centers[j] = Xs[m].mean(axis=0) if np.any(m) else Xs[rnd.randint(0, Xs.shape[0])]
        if np.linalg.norm(new_centers - centers_s) < 1e-4:
            centers_s = new_centers; labels = new_labels; break
        centers_s = new_centers; labels = new_labels
    centers = centers_s * sigma + mu
    def predict_grid(grid):
        Gs = (grid - mu) / sigma
        dists = np.stack([np.sum((Gs - centers_s[j])**2, axis=1) for j in range(k)], axis=1)
        return np.argmin(dists, axis=1)

# 3) Grupisanje cene: Low/Mid/High
bins = [1, 334, 667, 1001]
labels_bins = ['Low', 'Mid', 'High']
price_group = np.array(labels_bins)[np.digitize(price, bins, right=False) - 1]

cluster_to_pricegrp = {}
for j in range(k):
    m = (labels == j)
    if np.any(m):
        vals, counts = np.unique(price_group[m], return_counts=True)
        cluster_to_pricegrp[j] = vals[np.argmax(counts)]
    else:
        cluster_to_pricegrp[j] = 'N/A'

# 4) Decision surface: x = Dan (20–25), y = Cena (1–1000)
x_min, x_max = 20.0, 25.0
y_min, y_max = 1.0, 1000.0
xx, yy = np.meshgrid(np.linspace(x_min, x_max, 400), np.linspace(y_min, y_max, 400))
grid = np.c_[xx.ravel(), yy.ravel()]
Z = predict_grid(grid).reshape(xx.shape)

colors = ['#3b82f6', '#ef4444', '#22c55e']  # plava, crvena, zelena
markers = {'Rainbow':'o','Curry':'s','Ketchup':'^','Icecream':'D'}

plt.figure(figsize=(10.2, 6.8))
plt.contourf(xx, yy, Z, levels=np.arange(-0.5, k+0.5, 1.0), alpha=0.22, colors=colors)
for j in range(k):
    m = (labels == j)
    plt.scatter(day_idx[m], price[m], s=30, alpha=0.95, c=colors[j], marker='o',
                label=f'Klaster {j} → {cluster_to_pricegrp[j]}')
for p in partije:
    m = (party == p)
    plt.scatter(day_idx[m], price[m], s=18, alpha=0.9, facecolors='none', edgecolors='black',
                marker=markers[p], label=f'Partija: {p}')
plt.scatter(centers[:,0], centers[:,1], s=160, c='black', marker='X', label='Centri')
plt.xlabel('Datum (dan u mesecu)'); plt.ylabel('Cena (1–1000)')
plt.title('K-Means mapa: 3 klastera po (Dan, Cena)\nVeza klastera sa grupom cene (Low/Mid/High) i pregled po partijama')
plt.grid(alpha=0.25)
plt.legend(bbox_to_anchor=(1.02, 1.0), loc='upper left')
plt.tight_layout()
plt.savefig('kmeans_day_price_map.png', dpi=150)
plt.show()

print('Dodali smo parametar Cena (1–1000) i napravili K-Means podelu na 3 klastera u prostoru (Dan, Cena), uz prikaz decision površine i centara.')
print('Klasteri su mapirani na grupe cene (Low/Mid/High) po većini, a markerima su prikazane i stvarne partije (Rainbow, Curry, Ketchup, Icecream).')
