
%matplotlib inline

import numpy as np
import matplotlib.pyplot as plt

# ==============================================
# 0) Generisanje podataka (1 feature) + outlier-i
# ==============================================
np.random.seed(42)
n = 150
X = np.random.uniform(-5, 5, size=(n, 1))
noise = np.random.normal(0, 2, size=n)
y = 3.0 * X[:, 0] + 5.0 + noise
# ubacimo outlier-e
y[np.random.choice(n, 5, replace=False)] += np.random.normal(20, 5, size=5)

# ==============================================
# Helper: metričke funkcije
# ==============================================
def r2_score_np(y_true, y_pred):
    ss_res = np.sum((y_true - y_pred)**2)
    ss_tot = np.sum((y_true - np.mean(y_true))**2)
    return 1 - ss_res/ss_tot

def mse_np(y_true, y_pred):
    return np.mean((y_true - y_pred)**2)

# ==============================================
# Pokušaj import-a sklearn (ako postoji -> koristi se)
# ==============================================
use_sklearn = True
try:
    from sklearn.linear_model import LinearRegression, Lasso, Ridge, ElasticNet
    from sklearn.preprocessing import StandardScaler
    from sklearn.pipeline import make_pipeline
except Exception:
    use_sklearn = False

# ==============================================
# 1) OLS
# ==============================================
if use_sklearn:
    ols = LinearRegression().fit(X, y)
    y_ols = ols.predict(X)
    r2_ols = r2_score_np(y, y_ols)
    mse_ols = mse_np(y, y_ols)
else:
    # OLS: beta = (X'X)^-1 X'y (sa intercept-om preko dodate kolone jedinica)
    X1 = np.c_[np.ones(n), X]
    beta = np.linalg.pinv(X1) @ y
    intercept_ols, coef_ols = beta[0], beta[1]
    y_ols = intercept_ols + coef_ols * X[:, 0]
    r2_ols = r2_score_np(y, y_ols)
    mse_ols = mse_np(y, y_ols)

# ==============================================
# 2) L1 (Lasso)
# ==============================================
alpha_l1 = 0.01  # slobodno menjaj po potrebi
if use_sklearn:
    lasso = make_pipeline(StandardScaler(), Lasso(alpha=alpha_l1, max_iter=10000))
    lasso.fit(X, y)
    y_lasso = lasso.predict(X)
    r2_lasso = r2_score_np(y, y_lasso)
    mse_lasso = mse_np(y, y_lasso)
    def predict_lasso_line(xx): return lasso.predict(xx)
else:
    # Fallback: koordinatna descenta za 1 feature (soft-thresholding)
    X_std = (X - X.mean(axis=0)) / X.std(axis=0, ddof=0)
    y_center = y - y.mean()
    x = X_std[:, 0]
    w = 0.0
    max_iter, tol = 2000, 1e-6
    for _ in range(max_iter):
        rho = np.dot(x, y_center - w*x) / len(x)
        if rho < -alpha_l1:
            w_new = (rho + alpha_l1) / (np.mean(x*x))
        elif rho > alpha_l1:
            w_new = (rho - alpha_l1) / (np.mean(x*x))
        else:
            w_new = 0.0
        if abs(w_new - w) < tol:
            w = w_new
            break
        w = w_new
    w_lasso_orig = w / (X.std(axis=0, ddof=0)[0])
    b_lasso_orig = y.mean() - w_lasso_orig * X.mean(axis=0)[0]
    y_lasso = b_lasso_orig + w_lasso_orig * X[:, 0]
    r2_lasso = r2_score_np(y, y_lasso)
    mse_lasso = mse_np(y, y_lasso)
    def predict_lasso_line(xx): return b_lasso_orig + w_lasso_orig * xx[:, 0]

# ==============================================
# 3) L2 (Ridge)
# ==============================================
alpha_l2 = 1.0   # slobodno menjaj po potrebi
if use_sklearn:
    ridge = make_pipeline(StandardScaler(), Ridge(alpha=alpha_l2))
    ridge.fit(X, y)
    y_ridge = ridge.predict(X)
    r2_ridge = r2_score_np(y, y_ridge)
    mse_ridge = mse_np(y, y_ridge)
    def predict_ridge_line(xx): return ridge.predict(xx)
else:
    # Fallback: ridge sa intercept-om (centiranje X i y)
    Xm = X.mean(axis=0)[0]
    Ym = y.mean()
    Xc = X[:, 0] - Xm
    yc = y - Ym
    # w_ridge = (Xc' yc) / (Xc' Xc + lambda)
    denom = np.dot(Xc, Xc) + alpha_l2
    w_ridge = np.dot(Xc, yc) / denom
    b_ridge = Ym - w_ridge * Xm
    y_ridge = b_ridge + w_ridge * X[:, 0]
    r2_ridge = r2_score_np(y, y_ridge)
    mse_ridge = mse_np(y, y_ridge)
    def predict_ridge_line(xx): return b_ridge + w_ridge * xx[:, 0]

# ==============================================
# 4) Elastic Net (kombinacija L1 i L2)
# ==============================================
alpha_en = 0.01   # ukupna jačina regularizacije
l1_ratio = 0.5    # udeo L1; ostatak je L2 (0..1)
if use_sklearn:
    en = make_pipeline(StandardScaler(), ElasticNet(alpha=alpha_en, l1_ratio=l1_ratio, max_iter=10000))
    en.fit(X, y)
    y_en = en.predict(X)
    r2_en = r2_score_np(y, y_en)
    mse_en = mse_np(y, y_en)
    def predict_en_line(xx): return en.predict(xx)
else:
    # Fallback: koordinatna descenta za 1 feature sa L1+L2
    X_std = (X - X.mean(axis=0)) / X.std(axis=0, ddof=0)
    y_center = y - y.mean()
    x = X_std[:, 0]
    w = 0.0
    max_iter, tol = 2000, 1e-6
    lam1 = alpha_en * l1_ratio
    lam2 = alpha_en * (1.0 - l1_ratio)
    for _ in range(max_iter):
        rho = np.dot(x, y_center - w*x) / len(x)
        # Soft-thresholding uz dodatno skupljanje od L2 u imenitelju
        m = np.mean(x*x) + lam2
        if rho < -lam1:
            w_new = (rho + lam1) / m
        elif rho > lam1:
            w_new = (rho - lam1) / m
        else:
            w_new = 0.0
        if abs(w_new - w) < tol:
            w = w_new
            break
        w = w_new
    w_en_orig = w / (X.std(axis=0, ddof=0)[0])
    b_en_orig = y.mean() - w_en_orig * X.mean(axis=0)[0]
    y_en = b_en_orig + w_en_orig * X[:, 0]
    r2_en = r2_score_np(y, y_en)
    mse_en = mse_np(y, y_en)
    def predict_en_line(xx): return b_en_orig + w_en_orig * xx[:, 0]

# ==============================================
# 5) Plot: podaci + OLS + Lasso + Ridge + Elastic
# ==============================================
plt.figure(figsize=(10, 6))
plt.scatter(X[:, 0], y, s=30, alpha=0.6, label='Podaci')
xx = np.linspace(X.min(), X.max(), 200).reshape(-1, 1)

# OLS
if use_sklearn:
    plt.plot(xx[:, 0], ols.predict(xx), color='tab:blue', linewidth=2,
             label=f'OLS (R²={r2_ols:.3f}, MSE={mse_ols:.2f})')
else:
    plt.plot(xx[:, 0], (intercept_ols + coef_ols * xx[:, 0]), color='tab:blue', linewidth=2,
             label=f'OLS (R²={r2_ols:.3f}, MSE={mse_ols:.2f})')

# Lasso
plt.plot(xx[:, 0], predict_lasso_line(xx), color='tab:orange', linewidth=2,
         label=f'Lasso L1 (α={alpha_l1}, R²={r2_lasso:.3f}, MSE={mse_lasso:.2f})')

# Ridge
plt.plot(xx[:, 0], predict_ridge_line(xx), color='tab:green', linewidth=2,
         label=f'Ridge L2 (α={alpha_l2}, R²={r2_ridge:.3f}, MSE={mse_ridge:.2f})')

# Elastic Net
plt.plot(xx[:, 0], predict_en_line(xx), color='tab:purple', linewidth=2,
         label=f'Elastic Net (α={alpha_en}, L1 ratio={l1_ratio}, R²={r2_en:.3f}, MSE={mse_en:.2f})')

plt.title('OLS vs L1 (Lasso) vs L2 (Ridge) vs Elastic Net')
plt.xlabel('X'); plt.ylabel('y'); plt.legend(); plt.tight_layout()
plt.show()

# ==============================================
# 6) Završne dve rečenice (tačno dve)
# ==============================================
print('Kreirali smo sintetičke podatke i uporedili OLS, L1 (Lasso), L2 (Ridge) i Elastic Net na istom grafiku i metrikama.')
print('L1 skuplja koeficijent ka nuli (sparsnije rešenje), L2 ga stabilizuje i smanjuje varijansu, dok Elastic Net kombinuje obe kazne i balansira između robusnosti i sparznosti.')
